cmake_minimum_required(VERSION 3.28.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(minew_radio_diagnostic)
if(NOT RADIO_ROLE MATCHES "^(tx|rx)$")
  message(FATAL_ERROR "RADIO_ROLE must be tx or rx")
endif()
target_compile_definitions(app PRIVATE RADIO_TX=$<STREQUAL:${RADIO_ROLE},tx>)
if(CONFIG_MINEW_RADIO_RELIABLE)
  target_sources(app PRIVATE src/reliable.c)
else()
  target_sources(app PRIVATE src/main.c)
endif()
if(CONFIG_MINEW_RADIO_CRYPTO_PROBE)
  target_sources(app PRIVATE src/crypto_probe.c)
endif()
if(CONFIG_MINEW_RADIO_DTLS_PROBE)
  target_sources(app PRIVATE src/dtls_probe.c)
endif()
if(CONFIG_MINEW_RADIO_SECURE)
  zephyr_include_directories(src)
  target_sources(app PRIVATE src/secure_dtls.c src/secure_pair.c)
endif()
if(RADIO_ROLE STREQUAL "rx")
  include(${ZEPHYR_BASE}/samples/subsys/usb/common/common.cmake)
  target_sources(app PRIVATE src/usb.c)
endif()
